def test_search_inventory_and_get_rates_agree_and_hide_cost():
    ctx = Ctx()
    res = advisor_tools.run_tool(ctx, "search_inventory", {"destination": "kyoto-japan", "type": "stay", "limit": 2})

    assert advisor_tools.run_tool(ctx, "get_rates", {"item_id": "item_9999"})["error"].startswith("No such item_id")

@pytest.mark.parametrize("dest", ["kyoto-japan", "kerala-india", "iceland", "bali-indonesia", "jaipur-india"])
def test_optimiser_is_deterministic_and_every_price_traces(dest):
    c = {"destination": dest, "nights": 4, "travellers": 3, "budget_input": {"amount": 4000, "currency": "USD"}}
    a, b = optimiser.optimise(dict(c)), optimiser.optimise(dict(c))
    assert a == b
    assert optimiser.trace_prices(a) == []
    assert a["within_budget"] and a["totals"]["sell_usd"] <= a["budget_usd"]
    assert len({l["item_id"] for l in a["lines"]}) == len(a["lines"])


def test_knapsack_never_loses_to_greedy_and_never_overspends():
    items = [{"id": f"i{n}", "type": "experience"} for n in range(6)]
    cands = [(3.0, 60.0, items[0]), (2.0, 40.0, items[1]), (2.0, 40.0, items[2]),
             (1.0, 30.0, items[3]), (0.9, 25.0, items[4]), (0.5, 10.0, items[5])]
    picked = optimiser._knapsack(cands, slots=3, budget_left=80.0, max_guides=1)
    cost = sum(c for _, c, i in cands if i in picked)
    value = sum(s for s, _, i in cands if i in picked)
    assert cost <= 80.0 and len(picked) <= 3 and value >= 4.0   # items 1+2 beat item 0 alone

def test_responsible_weighting_only_uses_verified_scores(monkeypatch):
    plain = optimiser.optimise(dict(TRIP))
    weighted = optimiser.optimise({**TRIP, "prefer_responsible": True})
    assert weighted["optimiser"]["responsible_weighted"] and not plain["optimiser"]["responsible_weighted"]

    def avg(p):
        s = [optimiser._verified_score(next(i for i in inv.state()["items"] if i["id"] == l["item_id"])) for l in p["lines"]]
        return sum(s) / len(s)
    assert avg(weighted) >= avg(plain)
    monkeypatch.setattr("app.services.responsible.supplier_summary", lambda sid: {"score": None})
    assert optimiser._verified_score({"supplier_id": "x"}) == 0.0   # no verified claim, no boost

def parse_sse(text):
    out = []
    for block in text.strip().split("\n\n"):
        ev, data = block.split("\n", 1)
        out.append((ev.removeprefix("event: "), json.loads(data.removeprefix("data: "))))
    return out


def test_stream_emits_steps_then_tokens_then_done(gemini):
    g = gemini(call("optimise_package", destination="kyoto-japan", nights=3, travellers=2), say("A lovely Kyoto plan for you."))
    r = client.post("/api/advisor/stream", json={"message": "3 nights in Kyoto for 2"})
    assert r.headers["content-type"].startswith("text/event-stream")
    ev = parse_sse(r.text)
    names = [e for e, _ in ev]
    assert names[0] == "status" and "step" in names and names[-1] == "done"
    assert names.index("step") < names.index("token") < names.index("done")
    done = ev[-1][1]
    assert "".join(d["text"] for e, d in ev if e == "token") == done["reply"] == "A lovely Kyoto plan for you."
    assert done["proposal"]["destination"] == "kyoto-japan" and done["agent"]["steps"][0]["tool"] == "optimise_package"


def test_stream_without_a_key_still_streams_the_rule_based_reply():
    ev = parse_sse(client.post("/api/advisor/stream", json={"message": "4 nights in Kerala for 3"}).text)
    assert ev[-1][0] == "done" and ev[-1][1]["agent"]["mode"] == "rules" and any(e == "token" for e, _ in ev)
